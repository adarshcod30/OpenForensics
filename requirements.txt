# Serving / app dependencies. Kept lean: this is what Streamlit Community
# Cloud and Hugging Face Spaces install, and both are memory-constrained.
# Training extras live in requirements-train.txt.

# TF 2.19 ships Keras 3. The published checkpoints are written by Keras 3.12,
# so a Keras 2 install (tensorflow < 2.16) cannot deserialise them at all --
# it fails on `keras.src.models.functional`. Do not lower this pin.
tensorflow==2.19.1
keras>=3.12,<3.13

numpy>=2.0,<3
Pillow>=10.0
matplotlib>=3.9          # colormap for the Grad-CAM overlay
streamlit>=1.40
huggingface-hub>=0.23    # fetches model.keras at first run
